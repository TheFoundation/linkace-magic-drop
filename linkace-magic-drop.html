#!/usr/bin/env python3 """ wwwoffle2warc.py — Exportiert den WWWOFFLE-Cache als WARC-Datei. Voraussetzungen: - wwwoffle-tools müssen installiert sein (wwwoffle-ls, wwwoffle-read), diese liegen normalerweise unter /usr/local/bin oder /usr/bin nach "make install" bzw. sind Teil des wwwoffle-Pakets. - Skript muss dort laufen, wo der Spool-Ordner (default: /var/spool/wwwoffle) lesbar ist bzw. wo wwwoffled selbst läuft. - pip install warcio --break-system-packages Benutzung: python3 wwwoffle2warc.py [ausgabe.warc.gz] [--spool /var/spool/wwwoffle] """ import argparse import io import os import subprocess import sys from warcio.statusandheaders import StatusAndHeadersParser from warcio.warcwriter import WARCWriter def list_urls(spool_dir: str) -> list[str]: """Durchsucht die Protokoll-Verzeichnisse und sammelt alle gecachten URLs.""" urls = [] for proto in ("http", "https", "ftp"): proto_dir = os.path.join(spool_dir, proto) if not os.path.isdir(proto_dir): continue for host in os.listdir(proto_dir): rel = f"{proto}/{host}" result = subprocess.run( ["wwwoffle-ls", rel], capture_output=True, text=True, check=False, ) if result.returncode != 0: print(f"Warnung: wwwoffle-ls {rel} fehlgeschlagen: " f"{result.stderr.strip()}", file=sys.stderr) continue for line in result.stdout.splitlines(): parts = line.split() if not parts: continue # Bei "ls -l"-artiger Ausgabe steht die URL im letzten Feld. candidate = parts[-1] if "://" in candidate: urls.append(candidate) return urls def read_raw_response(url: str) -> bytes: """Liest die vollständige gecachte HTTP-Antwort (Header + Body) roh aus.""" result = subprocess.run( ["wwwoffle-read", url], capture_output=True, check=False, ) if result.returncode != 0: print(f"Warnung: wwwoffle-read {url} fehlgeschlagen: " f"{result.stderr.decode(errors='replace').strip()}", file=sys.stderr) return b"" return result.stdout def main() -> None: ap = argparse.ArgumentParser() ap.add_argument("output", nargs="?", default="wwwoffle-export.warc.gz") ap.add_argument("--spool", default="/var/spool/wwwoffle") args = ap.parse_args() urls = list_urls(args.spool) print(f"{len(urls)} URLs im Cache gefunden.", file=sys.stderr) parser = StatusAndHeadersParser(["HTTP/1.0", "HTTP/1.1"]) written = 0 with open(args.output, "wb") as fh: writer = WARCWriter(fh, gzip=True) for url in urls: raw = read_raw_response(url) if not raw: continue stream = io.BytesIO(raw) try: http_headers = parser.parse(stream) except Exception as exc: print(f"Überspringe {url}: Header nicht parsbar ({exc})", file=sys.stderr) continue record = writer.create_warc_record( url, "response", payload=stream, http_headers=http_headers, ) writer.write_record(record) written += 1 print(f"{written} Records geschrieben nach {args.output}", file=sys.stderr) if __name__ == "__main__": main()
